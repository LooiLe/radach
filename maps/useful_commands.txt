
# ============================================
# FIRST-TIME SETUP (new developer, one-time)
# ============================================

# --- Windows (PowerShell) ---
# 1. Allow PowerShell to run scripts
Set-ExecutionPolicy -Scope CurrentUser -ExecutionPolicy RemoteSigned
# 2. Generate SSH key (skip if you already have one)
ssh-keygen -t rsa -b 4096
# 3. Copy your public key to the VPS (will ask for password once)
type $env:USERPROFILE\.ssh\id_rsa.pub | ssh root@93.127.194.132 "cat >> ~/.ssh/authorized_keys"

# --- Mac / Linux / WSL ---
# 1. Generate SSH key (skip if you already have one)
ssh-keygen -t rsa -b 4096
# 2. Copy your public key to the VPS (will ask for password once)
ssh-copy-id root@93.127.194.132
# 3. Make deploy script executable (one-time)
chmod +x deploy.sh


# ============================================
# DEPLOY (run from project root)
# ============================================

# Windows (PowerShell):
.\deploy.ps1              # deploy everything
.\deploy.ps1 -frontend    # frontend only
.\deploy.ps1 -backend     # backend only

# Mac / Linux / WSL:
./deploy.sh               # deploy everything
./deploy.sh frontend      # frontend only
./deploy.sh backend       # backend only
